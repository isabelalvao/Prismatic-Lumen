cmake_minimum_required(VERSION 3.24)
project(Fantasynt VERSION 0.12.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FANTASYNT_BUILD_APP "Build the SDL3/bgfx application" OFF)
include(CTest)
include(cmake/Dependencies.cmake)

    src/control/protocol.cpp

if(MSVC)
    target_compile_options(fantasynt_motifs PRIVATE /W4 /permissive- /utf-8)
endif()
if(MSVC)
    target_compile_options(fantasynt_control PRIVATE /W4 /permissive- /utf-8)
endif()

        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:fantasynt>/shaders"
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:fantasynt>/examples"
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:fantasynt>/audio"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets/opening" "$<TARGET_FILE_DIR:fantasynt>/opening"
        COMMAND ${CMAKE_COMMAND} "-DSOURCE=${CMAKE_SOURCE_DIR}/assets/visual-config.json"
            "-DDESTINATION=$<TARGET_FILE_DIR:fantasynt>/visual-config.json" -P "${CMAKE_SOURCE_DIR}/cmake/CopyVisualConfig.cmake"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/assets/visual-config.json" "$<TARGET_FILE_DIR:fantasynt>/visual-config.default.json"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/assets/visual-config-guide.md" "$<TARGET_FILE_DIR:fantasynt>/visual-config-guide.md"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/assets/audio/ChezNous_MASTER.wav" "$<TARGET_FILE_DIR:fantasynt>/audio"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/assets/audio/provenance.json" "$<TARGET_FILE_DIR:fantasynt>/audio"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:openxr_loader> "$<TARGET_FILE_DIR:fantasynt>"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FANTASYNT_SHADER_OUTPUTS} "$<TARGET_FILE_DIR:fantasynt>/shaders"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_SOURCE_DIR}/examples/calm.json"
            "${CMAKE_SOURCE_DIR}/examples/balanced.json"
            "${CMAKE_SOURCE_DIR}/examples/energetic.json" "$<TARGET_FILE_DIR:fantasynt>/examples"
        DEPENDS fantasynt_shaders openxr_loader ${FANTASYNT_SHADER_OUTPUTS} assets/audio/ChezNous_MASTER.wav assets/audio/provenance.json
            examples/calm.json examples/balanced.json examples/energetic.json
        VERBATIM)
    add_dependencies(fantasynt fantasynt_resources)
    if(BUILD_TESTING)

        add_dependencies(fantasynt_renderer_tests fantasynt_resources)

            "${CMAKE_BINARY_DIR}/shaders" "${CMAKE_SOURCE_DIR}/plan/m1-resize.png")
        set_tests_properties(renderer_resize PROPERTIES TIMEOUT 30)
